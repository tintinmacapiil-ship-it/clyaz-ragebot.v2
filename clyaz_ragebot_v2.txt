local function SaveOriginalSky()
    if OriginalSky then return end
    OriginalSky = {
        ClockTime      = Lighting.ClockTime,
        Brightness     = Lighting.Brightness,
        Ambient        = Lighting.Ambient,
        OutdoorAmbient = Lighting.OutdoorAmbient,
        FogColor       = Lighting.FogColor,
        FogEnd         = Lighting.FogEnd,
        FogStart       = Lighting.FogStart,
    }
    local sky = Lighting:FindFirstChildOfClass("Sky")
    if sky then
        OriginalSky.SkyboxBk = sky.SkyboxBk
        OriginalSky.SkyboxDn = sky.SkyboxDn
        OriginalSky.SkyboxFt = sky.SkyboxFt
        OriginalSky.SkyboxLf = sky.SkyboxLf
        OriginalSky.SkyboxRt = sky.SkyboxRt
        OriginalSky.SkyboxUp = sky.SkyboxUp
        OriginalSky.StarCount= sky.StarCount
    end
end

local function ApplySkybox(preset)
    SaveOriginalSky()
    local data = SkyboxPresets[preset]
    if not data then return end

    Lighting.ClockTime      = data.ClockTime      or Lighting.ClockTime
    Lighting.Brightness     = data.Brightness     or Lighting.Brightness
    Lighting.Ambient        = data.Ambient        or Lighting.Ambient
    Lighting.OutdoorAmbient = data.OutdoorAmbient or Lighting.OutdoorAmbient
    Lighting.FogColor       = data.FogColor       or Lighting.FogColor
    Lighting.FogEnd         = data.FogEnd         or Lighting.FogEnd
    Lighting.FogStart       = data.FogStart       or Lighting.FogStart

    local sky = Lighting:FindFirstChildOfClass("Sky") or Instance.new("Sky", Lighting)
    sky.SkyboxBk = data.SkyboxBk or sky.SkyboxBk
    sky.SkyboxDn = data.SkyboxDn or sky.SkyboxDn
    sky.SkyboxFt = data.SkyboxFt or sky.SkyboxFt
    sky.SkyboxLf = data.SkyboxLf or sky.SkyboxLf
    sky.SkyboxRt = data.SkyboxRt or sky.SkyboxRt
    sky.SkyboxUp = data.SkyboxUp or sky.SkyboxUp
    if data.StarCount then sky.StarCount = data.StarCount end

    -- Post-process for Nebula: add BloomEffect
    if preset == "Nebula" then
        local bloom = Lighting:FindFirstChildOfClass("BloomEffect")
                   or Instance.new("BloomEffect", Lighting)
        bloom.Intensity  = 1.2
        bloom.Size       = 24
        bloom.Threshold  = 0.8
    end
end

local function ResetSkybox()
    if not OriginalSky then return end
    Lighting.ClockTime      = OriginalSky.ClockTime
    Lighting.Brightness     = OriginalSky.Brightness
    Lighting.Ambient        = OriginalSky.Ambient
    Lighting.OutdoorAmbient = OriginalSky.OutdoorAmbient
    Lighting.FogColor       = OriginalSky.FogColor
    Lighting.FogEnd         = OriginalSky.FogEnd
    Lighting.FogStart       = OriginalSky.FogStart
    local sky = Lighting:FindFirstChildOfClass("Sky")
    if sky and OriginalSky.SkyboxBk then
        sky.SkyboxBk = OriginalSky.SkyboxBk
        sky.SkyboxDn = OriginalSky.SkyboxDn
        sky.SkyboxFt = OriginalSky.SkyboxFt
        sky.SkyboxLf = OriginalSky.SkyboxLf
        sky.SkyboxRt = OriginalSky.SkyboxRt
        sky.SkyboxUp = OriginalSky.SkyboxUp
        sky.StarCount= OriginalSky.StarCount
    end
    local bloom = Lighting:FindFirstChildOfClass("BloomEffect")
    if bloom then bloom:Destroy() end
end

-- ═══════════════════════════════════════════════════════════════
--  STRETCH SCREEN (CFrame matrix column squish)
-- ═══════════════════════════════════════════════════════════════
local function EnableStretch()
    if StretchConn then StretchConn:Disconnect() end
    StretchConn = RunService.RenderStepped:Connect(function()
        if not Config.StretchEnabled then return end
        local cf = Camera.CFrame
        local ratio = Config.StretchRatio
        Camera.CFrame = CFrame.fromMatrix(
            cf.Position,
            cf.XVector * ratio,
            cf.YVector,
            cf.ZVector
        )
    end)
end

local function DisableStretch()
    if StretchConn then StretchConn:Disconnect(); StretchConn = nil end
    Camera.FieldOfView = 70
end

local function IsSameTeam(player)
    if not LocalPlayer.Team or not player.Team then return false end
    return LocalPlayer.Team == player.Team
end

local function IsVisible(part)
    local char    = LocalPlayer.Character
    local origin  = Camera.CFrame.Position
    local dir     = part.Position - origin
    local rp      = RaycastParams.new()
    rp.FilterDescendantsInstances = {char or {}, part.Parent}
    rp.FilterType = Enum.RaycastFilterType.Exclude
    local result  = Workspace:Raycast(origin, dir.Unit * dir.Magnitude, rp)
    return result == nil
end

local function GetBestPart(char)
    local partOrder = {
        Config.TargetPart ~= "Auto" and Config.TargetPart or nil,
        "Head", "HumanoidRootPart", "UpperTorso", "LowerTorso",
        "RightUpperArm", "LeftUpperArm",
    }
    for _, pname in ipairs(partOrder) do
        if pname then
            local p = char:FindFirstChild(pname)
            if p then
                if not Config.WallCheck or IsVisible(p) then
                    return p
                end
            end
        end
    end
    return nil
end

-- ═══════════════════════════════════════════════════════════════
--  BACKTRACK SYSTEM (ring buffer, 180ms history)
-- ═══════════════════════════════════════════════════════════════
local function StoreBacktrackFrame()
    if not Config.BacktrackEnabled then return end
    local now = tick()
    for _, player in ipairs(Players:GetPlayers()) do
        if player == LocalPlayer then continue end
        local char = player.Character
        if not char then continue end
        if not BacktrackCache[player] then BacktrackCache[player] = {} end
        local cache = BacktrackCache[player]

        local parts = {"Head","HumanoidRootPart","UpperTorso"}
        local frame = {t = now, positions = {}}
        for _, pname in ipairs(parts) do
            local p = char:FindFirstChild(pname)
            if p then frame.positions[pname] = p.Position end
        end
        table.insert(cache, 1, frame)

        -- Prune frames older than BacktrackDuration
        while #cache > 0 and (now - cache[#cache].t) > Config.BacktrackDuration do
            table.remove(cache)
        end
    end
end

local function GetBacktrackPos(player)
    local cache = BacktrackCache[player]
    if not cache or #cache == 0 then return nil end
    local now = tick()
    -- Newest to oldest, find first frame where head was visible
    for _, frame in ipairs(cache) do
        if (now - frame.t) <= Config.BacktrackDuration then
            local pos = frame.positions[Config.TargetPart] or frame.positions["Head"]
            if pos then
                -- Quick visibility check at cached position (approximate)
                local origin = Camera.CFrame.Position
                local dir    = pos - origin
                local rp     = RaycastParams.new()
                rp.FilterDescendantsInstances = {LocalPlayer.Character or {}}
                rp.FilterType = Enum.RaycastFilterType.Exclude
                local res = Workspace:Raycast(origin, dir.Unit * math.min(dir.Magnitude, 1000), rp)
                if not res then return pos end
            end
        end
    end
    return nil
end

-- ═══════════════════════════════════════════════════════════════
--  TARGET ACQUISITION
-- ═══════════════════════════════════════════════════════════════
local function FindTarget()
    if not Config.SilentAim and not Config.AutoFire then
        AimTarget = nil; AimTargetPart = nil; AimTargetPos = nil
        return
    end

    local mousePos = UserInputService:GetMouseLocation()
    local bestDist = math.huge
    local bestPlayer, bestPart = nil, nil

    for _, player in ipairs(Players:GetPlayers()) do
        if player == LocalPlayer then continue end
        if Config.TeamCheck and IsSameTeam(player) then continue end

        local char = player.Character
        if not char then continue end
        local hum  = char:FindFirstChildOfClass("Humanoid")
        if not hum or hum.Health <= 0 then continue end

        local part = GetBestPart(char)
        if not part then continue end

        local sp, onScreen = Camera:WorldToViewportPoint(part.Position)
        if not onScreen or sp.Z < 0 then continue end

        local screenDist = (Vector2.new(sp.X, sp.Y) - mousePos).Magnitude
        if screenDist <= Config.FOVRadius and screenDist < bestDist then
            bestDist   = screenDist
            bestPlayer = player
            bestPart   = part
        end
    end

    AimTarget    = bestPlayer
    AimTargetPart= bestPart

    if bestPlayer then
        -- Try backtrack first, fall back to real position
        if Config.BacktrackEnabled then
            local btPos = GetBacktrackPos(bestPlayer)
            AimTargetPos = btPos or (bestPart and bestPart.Position)
        else
            AimTargetPos = bestPart and bestPart.Position
        end
    else
        AimTargetPos = nil
    end
end

-- ═══════════════════════════════════════════════════════════════
--  FIRE ENGINE
-- ═══════════════════════════════════════════════════════════════
local function FireSingle()
    mouse1press()
    task.wait(Config.FireDelay)
    mouse1release()
end

local function FireDoubleTap()
    -- Two shots in rapid succession — abuses lag comp window
    for i = 1, Config.ShotsPerCycle do
        mouse1press()
        task.wait(0.013)
        mouse1release()
        task.wait(0.013)
    end
end

local FireEngineActive = false
local function RunFireEngine()
    if FireEngineActive then return end
    FireEngineActive = true
    task.spawn(function()
        while Config.AutoFire and AimTarget do
            if Config.DoubleTap then
                FireDoubleTap()
            else
                FireSingle()
            end
            task.wait(Config.FireDelay)
        end
        FireEngineActive = false
    end)
end

-- ═══════════════════════════════════════════════════════════════
--  VOID / FAKE POSITION  (K8X Flicker — ported from w7w7)
-- ═══════════════════════════════════════════════════════════════
task.spawn(function()
    while true do
        if Config.VoidEnabled then
            pcall(function()
                local char = LocalPlayer.Character
                local hrp  = char and char:FindFirstChild("HumanoidRootPart")
                if hrp then
                    -- Save real position before first void
                    if not VoidLastPos then VoidLastPos = hrp.CFrame end

                    -- K8X core: massive alternating random coords
                    local min_v, max_v = 10000000000, 100000000000
                    local rx = math.random(min_v, max_v)
                    local ry = math.random(min_v, max_v)
                    local rz = math.random(min_v, max_v)
                    if not VoidFlip then rx=-rx; ry=-ry; rz=-rz end
                    VoidFlip = not VoidFlip

                    hrp.CFrame = CFrame.new(rx, ry, rz)
                    hrp.AssemblyLinearVelocity = Vector3.zero
                end
            end)
        else
            VoidFlip = true
        end
        task.wait(Config.VoidRate)
    end
end)

-- ═══════════════════════════════════════════════════════════════
--  ENEMY KICK + PROJECTILE DEFLECTION  (ported from w7w7)
-- ═══════════════════════════════════════════════════════════════
RunService.Heartbeat:Connect(function()
    -- Kick: void all enemies and their projectiles
    if Config.KickEnabled then
        pcall(function()
            for _, player in ipairs(Players:GetPlayers()) do
                if player ~= LocalPlayer and player.Character then
                    local hrp = player.Character:FindFirstChild("HumanoidRootPart")
                    if hrp then
                        hrp.CFrame = CFrame.new(VOID_KICK)
                        hrp.AssemblyLinearVelocity = Vector3.zero
                    end
                end
            end
        end)
        pcall(function()
            for proj in pairs(Projectiles) do
                if proj and proj.Parent then
                    proj.CFrame = CFrame.new(VOID_KICK)
                    proj.AssemblyLinearVelocity = Vector3.zero
                else
                    Projectiles[proj] = nil
                end
            end
        end)
    end
end)

-- ═══════════════════════════════════════════════════════════════
--  TP RAGE  (cycle teleport to players — from w7w7)
-- ═══════════════════════════════════════════════════════════════
task.spawn(function()
    while true do
        if Config.TPEnabled then
            pcall(function()
                local myChar = LocalPlayer.Character
                local myHRP  = myChar and myChar:FindFirstChild("HumanoidRootPart")
                if not myHRP then return end

                local targets = {}
                for _, p in ipairs(Players:GetPlayers()) do
                    if p ~= LocalPlayer then
                        if not Config.TeamCheck or not IsSameTeam(p) then
                            if p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
                                table.insert(targets, p)
                            end
                        end
                    end
                end

                for _, target in ipairs(targets) do
                    if not Config.TPEnabled then break end
                    local tHRP = target.Character and target.Character:FindFirstChild("HumanoidRootPart")
                    if tHRP then
                        myHRP.CFrame = tHRP.CFrame * CFrame.new(0, 0, 2.5)
                        myHRP.AssemblyLinearVelocity = Vector3.zero
                        task.wait(Config.TPInterval)
                    end
                end
            end)
        end
        task.wait(0.4)
    end
end)

-- ═══════════════════════════════════════════════════════════════
--  MASTER RENDER LOOP
-- ═══════════════════════════════════════════════════════════════
local frameCount = 0
RunService.RenderStepped:Connect(function(dt)
    frameCount = frameCount + 1

    -- Target scan every frame
    FindTarget()

    -- Backtrack storage every frame
    if Config.BacktrackEnabled then
        StoreBacktrackFrame()
    end

    -- Auto fire when target locked
    if Config.AutoFire and AimTarget then
        RunFireEngine()
    end

    -- FOV circle
    if Config.ShowFOVCircle then
        FovCircle.Visible  = true
        FovCircle.Position = UserInputService:GetMouseLocation()
        FovCircle.Radius   = Config.FOVRadius
        FovCircle.Color    = AimTarget
            and Color3.fromRGB(255, 60, 60)
            or  Color3.fromRGB(180, 180, 200)
    else
        FovCircle.Visible = false
    end

    -- Target HUD
    if Config.ShowTargetHUD and AimTarget and AimTargetPart then
        local char   = AimTarget.Character
        local hum    = char and char:FindFirstChildOfClass("Humanoid")
        local hp     = hum and hum.Health    or 0
        local maxHp  = hum and hum.MaxHealth or 100
        local hpPct  = math.clamp(hp / math.max(maxHp, 1), 0, 1)
        local hpColor = Color3.fromRGB(
            math.floor(255 * (1 - hpPct)),
            math.floor(255 * hpPct),
            50
        )

        local head  = char:FindFirstChild("Head") or AimTargetPart
        local root  = char:FindFirstChild("HumanoidRootPart") or AimTargetPart
        local topPos= head.Position + Vector3.new(0, 0.7, 0)
        local botPos= root.Position - Vector3.new(0, 2.8, 0)

        local topSP, topOn  = Camera:WorldToViewportPoint(topPos)
        local botSP, botOn  = Camera:WorldToViewportPoint(botPos)
        local vp            = Camera.ViewportSize
        local centerX       = (topSP.X + botSP.X) * 0.5

        if topOn or botOn then
            local minY   = math.min(topSP.Y, botSP.Y)
            local maxY   = math.max(topSP.Y, botSP.Y)
            local height = maxY - minY
            local width  = height * 0.5
            local bx     = centerX - width * 0.5

            HUD.Box.Visible  = true
            HUD.Box.Position = Vector2.new(bx, minY)
            HUD.Box.Size     = Vector2.new(width, height)
            HUD.Box.Color    = Color3.fromRGB(255, 60, 60)

            HUD.Name.Visible  = true
            HUD.Name.Position = Vector2.new(centerX, minY - 16)
            HUD.Name.Text     = AimTarget.Name
            HUD.Name.Color    = Color3.new(1, 1, 1)

            local barW = width - 4
            local barH = 4
            local barX = bx + 2
            local barY = maxY + 4
            HUD.HPBg.Visible  = true
            HUD.HPBg.Position = Vector2.new(barX, barY)
            HUD.HPBg.Size     = Vector2.new(barW, barH)

            HUD.HPFill.Visible  = true
            HUD.HPFill.Position = Vector2.new(barX, barY)
            HUD.HPFill.Size     = Vector2.new(barW * hpPct, barH)
            HUD.HPFill.Color    = hpColor

local function MkSlider(labelText, minV, maxV, defaultV, onChange, isFloat)
    local row = Instance.new("Frame", ScrollArea)
    row.Size             = UDim2.new(1, -16, 0, 48)
    row.BackgroundColor3 = CL.ELEM
    row.BorderSizePixel  = 0  -- sharp
    row.LayoutOrder      = NextOrder()
    MkStroke(row, CL.BORDER, 1)

    local lbl = Instance.new("TextLabel", row)
    lbl.Size            = UDim2.new(0.65, 0, 0, 20)
    lbl.Position        = UDim2.fromOffset(10, 4)
    lbl.BackgroundTransparency = 1
    lbl.Text            = labelText
    lbl.TextColor3      = CL.TEXT
    lbl.TextSize        = 12
    lbl.Font            = Enum.Font.GothamBold
    lbl.TextXAlignment  = Enum.TextXAlignment.Left
    lbl.BorderSizePixel = 0

    local valLbl = Instance.new("TextLabel", row)
    valLbl.Size            = UDim2.new(0.35, -10, 0, 20)
    valLbl.Position        = UDim2.new(0.65, 0, 0, 4)
    valLbl.BackgroundTransparency = 1
    valLbl.Text            = isFloat and string.format("%.2f", defaultV) or tostring(defaultV)
    valLbl.TextColor3      = CL.ACC
    valLbl.TextSize        = 11
    valLbl.Font            = Enum.Font.GothamBold
    valLbl.TextXAlignment  = Enum.TextXAlignment.Right
    valLbl.BorderSizePixel = 0

    local track = Instance.new("Frame", row)
    track.Size             = UDim2.new(1, -20, 0, 4)
    track.Position         = UDim2.fromOffset(10, 32)
    track.BackgroundColor3 = CL.ELEM2
    track.BorderSizePixel  = 0  -- sharp

    local fillPct = (defaultV - minV) / (maxV - minV)
    local fill = Instance.new("Frame", track)
    fill.Size             = UDim2.new(fillPct, 0, 1, 0)
    fill.BackgroundColor3 = CL.ACC
    fill.BorderSizePixel  = 0

    local thumb = Instance.new("Frame", track)
    thumb.Size             = UDim2.fromOffset(10, 14)
    thumb.Position         = UDim2.new(fillPct, -5, 0.5, -7)
    thumb.BackgroundColor3 = CL.ACCLIT
    thumb.BorderSizePixel  = 0  -- sharp

    local curVal = defaultV
    local dragging = false

    local function Update(px)
        local abs = track.AbsolutePosition
        local sz  = track.AbsoluteSize
        local pct = math.clamp((px - abs.X) / sz.X, 0, 1)
        if isFloat then
            curVal = math.floor((minV + (maxV-minV)*pct) * 100 + 0.5) / 100
        else
            curVal = math.round(minV + (maxV-minV)*pct)
        end
        fill.Size      = UDim2.new(pct, 0, 1, 0)
        thumb.Position = UDim2.new(pct, -5, 0.5, -7)
        valLbl.Text    = isFloat and string.format("%.2f", curVal) or tostring(curVal)
        if onChange then onChange(curVal) end
    end

    thumb.InputBegan:Connect(function(inp)
        if inp.UserInputType == Enum.UserInputType.MouseButton1 then dragging=true end
    end)
    track.InputBegan:Connect(function(inp)
        if inp.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging=true; Update(inp.Position.X)
        end
    end)
    UserInputService.InputEnded:Connect(function(inp)
        if inp.UserInputType == Enum.UserInputType.MouseButton1 then dragging=false end
    end)
    UserInputService.InputChanged:Connect(function(inp)
        if dragging and inp.UserInputType == Enum.UserInputType.MouseMovement then
            Update(inp.Position.X)
        end
    end)

    row.MouseEnter:Connect(function() Tw(row, {BackgroundColor3=CL.ELEM2}) end)
    row.MouseLeave:Connect(function() Tw(row, {BackgroundColor3=CL.ELEM}) end)
end

    dropBtn.Position         = UDim2.new(0.52, 0, 0.5, -12)
    dropBtn.BackgroundColor3 = CL.ELEM2
    dropBtn.BorderSizePixel  = 0  -- sharp
    dropBtn.Text             = default .. "  ▾"
    dropBtn.TextColor3       = CL.ACC
    dropBtn.TextSize         = 11
    dropBtn.Font             = Enum.Font.GothamBold
    dropBtn.TextXAlignment   = Enum.TextXAlignment.Center
    MkStroke(dropBtn, CL.BORDER, 1)

    local panel = Instance.new("Frame", GUI)
    panel.BackgroundColor3 = CL.PANEL
    panel.BorderSizePixel  = 0  -- sharp
    panel.Size             = UDim2.fromOffset(160, #options * 28 + 6)
    panel.ZIndex           = 50
    panel.Visible          = false
    MkStroke(panel, CL.ACC, 1)

    local pList = Instance.new("UIListLayout", panel)
    pList.SortOrder = Enum.SortOrder.LayoutOrder
    pList.Padding   = UDim.new(0, 1)
    local pPad = Instance.new("UIPadding", panel)
    pPad.PaddingTop=UDim.new(0,3); pPad.PaddingBottom=UDim.new(0,3)
    pPad.PaddingLeft=UDim.new(0,3); pPad.PaddingRight=UDim.new(0,3)

    local curSel = default
    for i, opt in ipairs(options) do
        local ob = Instance.new("TextButton", panel)
        ob.Size             = UDim2.new(1,-6,0,24)
        ob.BackgroundColor3 = opt==curSel and CL.ELEM2 or CL.ELEM
        ob.BorderSizePixel  = 0

            for _, c in ipairs(panel:GetChildren()) do
                if c:IsA("TextButton") then
                    local isSel = c.Text:gsub("  ▾","") == opt
                    Tw(c,{BackgroundColor3 = isSel and CL.ELEM2 or CL.ELEM, TextColor3 = isSel and CL.ACC or CL.TEXT})
                end
            end
            panel.Visible = false
            if onChange then onChange(opt) end
        end)
    end

    dropBtn.MouseButton1Click:Connect(function()
        if panel.Visible then
            panel.Visible = false
        else
            local ap = dropBtn.AbsolutePosition
            local as = dropBtn.AbsoluteSize
            panel.Position = UDim2.fromOffset(ap.X, ap.Y + as.Y + 2)
            panel.Visible  = true
        end
    end)
    dropBtn.MouseEnter:Connect(function() Tw(dropBtn, {BackgroundColor3=CL.ELEM}) end)
    dropBtn.MouseLeave:Connect(function() Tw(dropBtn, {BackgroundColor3=CL.ELEM2}) end)
end
